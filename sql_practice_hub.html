-- ============================================
-- E-COMMERCE DATABASE SETUP
-- Compatible with PostgreSQL / MySQL / SQLite
-- ============================================

CREATE TABLE customers (
  customer_id INT PRIMARY KEY,
  name        VARCHAR(100),
  email       VARCHAR(150) UNIQUE,
  phone       VARCHAR(20),
  city        VARCHAR(80),
  state       VARCHAR(50),
  created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE categories (
  category_id        INT PRIMARY KEY,
  name               VARCHAR(100),
  parent_category_id INT REFERENCES categories(category_id)
);

CREATE TABLE products (
  product_id  INT PRIMARY KEY,
  category_id INT REFERENCES categories(category_id),
  name        VARCHAR(200),
  price       DECIMAL(10,2),
  stock_qty   INT DEFAULT 0,
  brand       VARCHAR(100),
  is_active   BOOLEAN DEFAULT TRUE
);

CREATE TABLE orders (
  order_id      INT PRIMARY KEY,
  customer_id   INT REFERENCES customers(customer_id),
  order_date    TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  status        VARCHAR(20) CHECK (status IN ('pending','confirmed','shipped','delivered','cancelled')),
  total_amount  DECIMAL(12,2),
  shipping_city VARCHAR(80),
  coupon_code   VARCHAR(30)
);

CREATE TABLE order_items (
  item_id    INT PRIMARY KEY,
  order_id   INT REFERENCES orders(order_id),
  product_id INT REFERENCES products(product_id),
  quantity   INT,
  unit_price DECIMAL(10,2),
  discount   DECIMAL(5,2) DEFAULT 0
);

CREATE TABLE reviews (
  review_id   INT PRIMARY KEY,
  product_id  INT REFERENCES products(product_id),
  customer_id INT REFERENCES customers(customer_id),
  rating      INT CHECK (rating BETWEEN 1 AND 5),
  comment     TEXT,
  reviewed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE shipments (
  shipment_id INT PRIMARY KEY,
  order_id    INT REFERENCES orders(order_id),
  carrier     VARCHAR(80),
  tracking_no VARCHAR(100),
  shipped_at  TIMESTAMP,
  delivered_at TIMESTAMP
);

-- ============================================
-- BANKING DATABASE SETUP
-- Compatible with PostgreSQL / MySQL / SQLite
-- ============================================

CREATE TABLE customers (
  customer_id INT PRIMARY KEY,
  name        VARCHAR(100),
  dob         DATE,
  email       VARCHAR(150) UNIQUE,
  city        VARCHAR(80),
  kyc_status  VARCHAR(20) CHECK (kyc_status IN ('pending','verified','rejected')),
  joined_at   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE branches (
  branch_id  INT PRIMARY KEY,
  name       VARCHAR(100),
  city       VARCHAR(80),
  state      VARCHAR(50),
  manager_id INT
);

CREATE TABLE accounts (
  account_id   INT PRIMARY KEY,
  customer_id  INT REFERENCES customers(customer_id),
  account_type VARCHAR(20) CHECK (account_type IN ('savings','current','fd','rd')),
  balance      DECIMAL(15,2) DEFAULT 0,
  currency     CHAR(3) DEFAULT 'INR',
  status       VARCHAR(20) CHECK (status IN ('active','frozen','closed')),
  opened_at    TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE transactions (
  txn_id     INT PRIMARY KEY,
  account_id INT REFERENCES accounts(account_id),
  txn_type   VARCHAR(20) CHECK (txn_type IN ('credit','debit')),
  amount     DECIMAL(12,2),
  channel    VARCHAR(20) CHECK (channel IN ('atm','upi','neft','imps','branch','online')),
  status     VARCHAR(20) CHECK (status IN ('success','failed','pending')),
  txn_at     TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE loans (
  loan_id       INT PRIMARY KEY,
  customer_id   INT REFERENCES customers(customer_id),
  loan_type     VARCHAR(30) CHECK (loan_type IN ('home','personal','vehicle','education')),
  principal     DECIMAL(15,2),
  interest_rate DECIMAL(5,2),
  tenure_months INT,
  disbursed_at  TIMESTAMP,
  status        VARCHAR(20) CHECK (status IN ('active','closed','defaulted'))
);

CREATE TABLE loan_payments (
  payment_id   INT PRIMARY KEY,
  loan_id      INT REFERENCES loans(loan_id),
  amount_paid  DECIMAL(12,2),
  payment_date DATE,
  is_overdue   BOOLEAN DEFAULT FALSE
);

CREATE TABLE fraud_flags (
  flag_id    INT PRIMARY KEY,
  txn_id     INT REFERENCES transactions(txn_id),
  account_id INT REFERENCES accounts(account_id),
  reason     VARCHAR(200),
  risk_score DECIMAL(4,2),
  flagged_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  resolved   BOOLEAN DEFAULT FALSE
);
